8PASTA = os.path.join(os.path.dirname(os.path.abspath(__file__)), "assets")

10janela = tk.Tk()

12tela = tk.Canvas(janela, width=768, height=480, highlightthickness=0)
13tela.pack()

15# Recortar um pedaco da imagem (x, y, largura, altura), com o dobro do tamanho
16def recorta(arquivo, x, y, largura, altura, espelhado=False):
17 folha = tk.PhotoImage(file=os.path.join(PASTA, arquivo))
18 quadro = tk.PhotoImage()
19 lado = -1 if espelhado else 1
20 quadro.tk.call(quadro, "copy", folha, "-from", x, y, x + largura, y + altura,
21 "-zoom", 2, 2, "-subsample", lado, 1)
22 return quadro
23 
24corre_dir = [recorta("raposa-corre.png", i * 33, 0, 33, 32) for i in range(6)]
25corre_esq = [recorta("raposa-corre.png", i * 33, 0, 33, 32, True) for i in range(6)]
26parada_dir = [recorta("raposa-parada.png", i * 33, 0, 33, 32) for i in range(4)]
27parada_esq = [recorta("raposa-parada.png", i * 33, 0, 33, 32, True) for i in range(4)]
28grama = recorta("chao.png", 16, 16, 16, 16)
29terra = recorta("chao.png", 16, 48, 16, 16)

40x, y, olhando, passo = 100, 352, "dir", 0
41raposa = tela.create_image(x, y, image=parada_dir[0], anchor="nw")

43teclas = set()
44janela.bind("<KeyPress>", lambda e: teclas.add(e.keysym))
45janela.bind("<KeyRelease>", lambda e: teclas.discard(e.keysym))
46janela.bind("<Escape>", lambda e: janela.destroy())

48# O game loop: ler teclas, mover, escolher o quadro, desenhar
49def quadro():
50 global x, olhando, passo
51 andando = False
52 if "Right" in teclas:
53 x, olhando, andando = x + 6, "dir", True
54 if "Left" in teclas:
55 x, olhando, andando = x - 6, "esq", True
56 x = max(0, min(x, 700))
57 passo = passo + 1
58 
59 if andando:
60 lista = corre_dir if olhando == "dir" else corre_esq
61 imagem = lista[passo % 6]
62 else:
63 lista = parada_dir if olhando == "dir" else parada_esq
64 imagem = lista[passo // 3 % 4]
65 
66 tela.coords(raposa, x, y)
67 tela.itemconfig(raposa, image=imagem)
68 janela.after(33, quadro)
69 
70quadro()
71janela.mainloop()
