25# Recortar um pedaco da imagem (x, y, largura, altura), com o dobro do tamanho
26def recorta(arquivo, x, y, largura, altura, espelhado=False):
27 folha = tk.PhotoImage(file=os.path.join(PASTA, arquivo))
28 quadro = tk.PhotoImage()
29 lado = -1 if espelhado else 1
30 quadro.tk.call(quadro, "copy", folha, "-from", x, y, x + largura, y + altura,
31 "-zoom", 2, 2, "-subsample", lado, 1)
32 return quadro

66# O game loop: ler teclas, mover, escolher o quadro, desenhar
67def quadro():
68 global x, y, vy, olhando, passo, pontos, cx, cy
69 andando = False
70 if "Right" in teclas:
71 x, olhando, andando = x + 6, "dir", True
72 if "Left" in teclas:
73 x, olhando, andando = x - 6, "esq", True
74 if "space" in teclas and y == CHAO:
75 vy = -18
76 vy = vy + 1.5
77 y = min(y + vy, CHAO)
78 x = max(0, min(x, 700))
79 passo = passo + 1
80 
81 if y < CHAO:
82 lista = pulo_dir if olhando == "dir" else pulo_esq
83 imagem = lista[0] if vy < 0 else lista[1]
84 elif andando:
85 lista = corre_dir if olhando == "dir" else corre_esq
86 imagem = lista[passo % 6]
87 else:
88 lista = parada_dir if olhando == "dir" else parada_esq
89 imagem = lista[passo // 3 % 4]
90 
91 if abs(x + 33 - cx) < 40 and abs(y + 32 - cy) < 40:
92 pontos = pontos + 1
93 cx, cy = random.randint(60, 700), random.randint(250, 390)
94 
95 tela.coords(raposa, x, y)
96 tela.itemconfig(raposa, image=imagem)
97 tela.coords(fruta, cx, cy)
98 tela.itemconfig(fruta, image=cereja[passo // 3 % 5])
99 tela.itemconfig(sombra, text=f"CEREJAS {pontos}")
100 tela.itemconfig(placar, text=f"CEREJAS {pontos}")
101 janela.after(33, quadro)
